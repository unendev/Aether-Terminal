# 1. DATA_MODEL.md - 实体与数据字典

## 1. Episode（单期归档切片）
```typescript
interface Episode {
  id: string;               // e.g. "EP001"
  title: string;            // e.g. "Genesis Protocol // 创世纪协议"
  logIndex: string;         // e.g. "LOG_2147_08_A"
  releaseDate: string;
  music: {
    bpm: number;
    key: string;
    vocalIdentity: string;  // e.g. "Aria-07"
    audioPath: string;
    stemsPath: string;
  };
  visual: {
    preset: string;         // 视觉预设，如 "terminal_void"
    dominantSymbols: string[]; // 核心字符，如 ["< >", "@", "#"]
    colorAccent?: string;   // 剧情关键色，如 "#FF3366" (若无则单色)
  };
  lore: {
    eventRef: string;       // 对应的历史事件 ID
    factionFocus: string;   // 关联阵营
    unresolvedClue: string; // 留下的悬念
  };
}
```

## 2. CharacterSymbolDNA（角色视觉与音色 DNA）
```typescript
interface CharacterSymbolDNA {
  id: string;               // e.g. "CHAR_OBSERVER_01"
  name: string;             // e.g. "Observer / 守望者"
  faction: string;          // e.g. "The Archive"
  symbolDNA: string;        // e.g. "< >" 或 "(@)"
  vocalProfile: string;     // 音色特征 (如: 空灵女性合成音、高频共鸣)
  firstAppearance: string;  // "EP001"
  status: "ACTIVE" | "CORRUPTED" | "ARCHIVED" | "UNKNOWN";
}
```

## 3. AudioReactiveParams（音频驱动视觉参数契约）
```typescript
interface AudioReactiveParams {
  bass: number;       // 0.0 - 1.0, 驱动重低音物理缩放、重力波纹
  mid: number;        // 0.0 - 1.0, 驱动结构网格、几何变形
  high: number;       // 0.0 - 1.0, 驱动噪点粒子、高频字符闪烁
  beat: boolean;      // 瞬态打击事件, 触发摄像机抖动与色散 Glitch
  vocalEnergy: number;// 0.0 - 1.0, 驱动中心符号亮度与情绪光环
}
```
